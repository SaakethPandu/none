# Example: Simple BMI Calculator
ask "Enter your height (m)" to height
ask "Enter your weight (kg)" to weight

set bmi to weight / (height * height)
say "Your BMI is " + bmi

if bmi > 25 then
    say "Consider more exercise!"
else
    say "You're doing great!"
end
